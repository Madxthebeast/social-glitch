static void safe_copy(char *d, size_t n, const char *s) {
    if (!s) s = "";
    if (n == 0) return;
    strncpy(d, s, n - 1);
    d[n - 1] = '\0';
}
static void json_escape(const char *src, char *dst, size_t n) {
    size_t j = 0;
    if (!src) src = "";
    for (size_t i = 0; src[i] && j + 2 < n; i++) {
        unsigned char c = (unsigned char)src[i];
        if (c == '"' || c == '\\') { dst[j++]='\\'; dst[j++]=(char)c; }
        else if (c == '\n') { dst[j++]='\\'; dst[j++]='n'; }
        else if (c == '\r') { dst[j++]='\\'; dst[j++]='r'; }
        else if (c == '\t') { dst[j++]='\\'; dst[j++]='t'; }
        else if (c >= 32) dst[j++] = (char)c;
    }
    dst[j] = '\0';
}

static void out_begin(char *o, size_t n, const char *data) {
    snprintf(o, n, "{\"ok\":true,\"data\":%s}", data ? data : "{}");

static void out_error(char *o, size_t n, const char *e) {
    char x[256]; json_escape(e, x, sizeof(x));
    snprintf(o, n, "{\"ok\":false,\"error\":\"%s\"}", x);
}
static const char *jstr(const char *json, const char *key, char *out, size_t cap) {
    char pat[128];
    snprintf(pat, sizeof(pat), "\"%s\"", key);
    const char *p = strstr(json, pat);
    if (!p) return NULL;
    p = strchr(p + strlen(pat), ':');
    if (!p) return NULL;
    p++;
    while (*p && isspace((unsigned char)*p)) p++;
    if (*p != '"') return NULL;
    p++;
    size_t j=0;
    while (*p && *p != '"' && j+1<cap) {
        if (*p=='\\' && p[1]) {
            p++;
            if (*p=='n') out[j++]='\n';
            else if (*p=='r') out[j++]='\r';
            else if (*p=='t') out[j++]='\t';
            else out[j++]=*p;
        } else out[j++]=*p;
        p++;
    }
    out[j]='\0';
    return out;
}

static int jint(const char *json, const char *key, int def) {
    char pat[128]; snprintf(pat,sizeof(pat),"\"%s\"",key);
    const char *p=strstr(json,pat); if(!p) return def;
    p=strchr(p+strlen(pat),':'); if(!p) return def; p++;
    while(*p && isspace((unsigned char)*p)) p++;
    return atoi(p);
}

static int jbool(const char *json, const char *key, int def) {
    char pat[128]; snprintf(pat,sizeof(pat),"\"%s\"",key);
    const char *p=strstr(json,pat); if(!p) return def;
    p=strchr(p+strlen(pat),':'); if(!p) return def; p++;
    while(*p && isspace((unsigned char)*p)) p++;
    if (!strncmp(p,"true",4)) return 1;
    if (!strncmp(p,"false",5)) return 0;
    return def;
}

static int uid_by_sgid(const char *sg) {

static int user_index(int id) {
    for(int i=0;i<nusers;i++) if(users[i].id==id) return i;
    return -1;
}
static User *user_ptr(int id) {
    int i=user_index(id); return i>=0 ? &users[i] : NULL;
}
static int username_exists(const char *u, int except) {
    for(int i=0;i<nusers;i++)
        if(users[i].id!=except && !strcasecmp(users[i].username,u)) return 1;
    return 0;
}
static int are_friends(int a,int b) {
    for(int i=0;i<nfriends;i++)
        if((friends_rel[i].a==a&&friends_rel[i].b==b)||(friends_rel[i].a==b&&friends_rel[i].b==a)) return 1;
    return 0;
}
static int is_blocked(int a,int b) {
    for(int i=0;i<nblocks;i++) if(blocks_rel[i].a==a&&blocks_rel[i].b==b) return 1;
    for(int i=0;i<nblocks;i++) if(blocks_rel[i].a==b&&blocks_rel[i].b==a) return 1;
    return 0;
}
static void add_friend(int a,int b) {
    if(!are_friends(a,b) && nfriends<MAX_REL) {
        friends_rel[nfriends].a=a; friends_rel[nfriends].b=b; nfriends++;
    }
}
static void remove_friend(int a,int b) {
    for(int i=0;i<nfriends;i++) if((friends_rel[i].a==a&&friends_rel[i].b==b)||(friends_rel[i].a==b&&friends_rel[i].b==a)) {
        friends_rel[i]=friends_rel[--nfriends]; i--;
    }
}
static int friend_count(int id) {
    int n=0; for(int i=0;i<nfriends;i++) if(friends_rel[i].a==id||friends_rel[i].b==id)n++; return n;
}
static int post_visible(Post *p,int me) {
    if(p->author==me) return 1;
    if(is_blocked(me,p->author)) return 0;
    if(p->visibility==0) return 1;
    if(p->visibility==1) return are_friends(me,p->author);
    return 0;
}

static int session_uid(const char *token) {
    if(!token || !*token) return 0;
    time_t now=time(NULL);
    for(int i=0;i<nsessions;i++)
        if(!strcmp(sessions[i].token,token)) {
            if(sessions[i].expires<now) { sessions[i]=sessions[--nsessions]; return 0; }
            return sessions[i].uid;
        }
    return 0;
}

static int strong_password(const char *p) {
    int lower=0, upper=0, digit=0, special=0; size_t n=strlen(p);
    if(n<10) return 0;
    for(size_t i=0;i<n;i++){ unsigned char c=(unsigned char)p[i];
        if(islower(c)) lower=1; else if(isupper(c)) upper=1; else if(isdigit(c)) digit=1; else if(!isspace(c)) special=1; }
    return lower&&upper&&digit&&special;
}

static int send_line(socket_t c,const char *s) {
    size_t len=strlen(s);char *x=(char*)malloc(len+2);if(!x)return 0;
    memcpy(x,s,len);x[len]='\n';x[len+1]=0;
    size_t sent=0;while(sent<len+1){int n=send(c,x+sent,(int)(len+1-sent),0);if(n<=0){free(x);return 0;}sent+=(size_t)n;}
    free(x);return 1;
}
static void handle_client(socket_t c) {
    char req[REQ_LEN],resp[RESP_LEN];size_t used=0;
    while(used<sizeof(req)-1){
        char ch;int n=recv(c,&ch,1,0);if(n<=0)break;
        if(ch=='\n')break;req[used++]=ch;
    }
    req[used]='\0';
    if(used){dispatch(req,resp,sizeof(resp));send_line(c,resp);}
}

int main(void) {
#ifdef _WIN32
    WSADATA w; if(WSAStartup(MAKEWORD(2,2),&w)!=0){fprintf(stderr,"WSAStartup failed\n");return 1;}
#endif
    seed();

    socket_t server=socket(AF_INET,SOCK_STREAM,0);
    if(server==INVALID_SOCKET){fprintf(stderr,"socket() failed\n");return 1;}

    int opt=1;
#ifdef _WIN32
    setsockopt(server,SOL_SOCKET,SO_REUSEADDR,(const char*)&opt,sizeof(opt));
#else
    setsockopt(server,SOL_SOCKET,SO_REUSEADDR,&opt,sizeof(opt));
#endif
    struct sockaddr_in addr;memset(&addr,0,sizeof(addr));
    addr.sin_family=AF_INET;addr.sin_addr.s_addr=inet_addr("127.0.0.1");addr.sin_port=htons(PORT);

    if(bind(server,(struct sockaddr*)&addr,sizeof(addr))==SOCKET_ERROR){
        fprintf(stderr,"Could not bind 127.0.0.1:%d\n",PORT);CLOSESOCK(server);return 1;
    }
    if(listen(server,16)==SOCKET_ERROR){
        fprintf(stderr,"listen() failed\n");CLOSESOCK(server);return 1;
    }
    printf("====================================\n");
    printf("       SOCIAL GLITCH BACKEND\n");
    printf("====================================\n");
    printf("Host: 127.0.0.1\nPort: %d\n\n",PORT);
    printf("Demo user : demo / Demo@1234\n");
    printf("Admin     : admin / Admin@1234\n\n");
    printf("Persistent database: %s\n",DB_FILE);
    printf("Waiting for Python frontend...\n");

    while(1){
        struct sockaddr_in client;
#ifdef _WIN32
        int clen=sizeof(client);
#else
        socklen_t clen=sizeof(client);
#endif
        socket_t c=accept(server,(struct sockaddr*)&client,&clen);
        if(c==INVALID_SOCKET)continue;
        handle_client(c);CLOSESOCK(c);
    }
    CLOSESOCK(server);
#ifdef _WIN32
    WSACleanup();
#endif
    return 0;
}
